<!DOCTYPE html>

<head>
    <meta charset="UTF-8">

    <style>
        body {
            display: flex;
            justify-content: center;
            align-items: center;
            min-height: 100vh;
            background-color: #f4f6fa;
            margin: 0;
            overflow: hidden;
            flex-direction: column;

            height: 4px;

            border-radius: 10px;
            overflow: hidden;
        }

        .progress-bar {
            width: 0%;
            height: 100%;
            background-color: #ffc107;
            border-radius: 10px;
            animation: load 4s linear forwards;
        }

        @keyframes load {
            0% { width: 0%; }
            100% { width: 100%; }
        }
    </style>
</head>
<body>
    <div class="progress-container">
        <div class="progress-bar"></div>
    </div>

    <script>
        const BASE_URL = 'https://api.archivehubs.com';
        window.onload = async function() {
            const response = await fetch(`${BASE_URL}/user`, {
                method: 'GET',
                headers: {
                    'Content-Type': 'application/json'
                },
                credentials: 'include',
            });

            if (response.ok){
                setTimeout(function() {
                    window.location.href = 'home.html';
                }, 4000); // 4000 milliseconds = 4 seconds
            } else {
                window.location.href = 'sign-in_account.html';
            }
        }
    </script>
</body>
</html>
